\section{Part 0 --- From Classical Control to Generative Policies}
%=====================================================================

\begin{frame}{Bài toán tổng quát: robot cần quyết định gì, ở đâu?}
\begin{itemize}
    \item \textbf{Bối cảnh thực tế}: robot tay máy (robot arm) trong nhà máy/kho hàng/nhà bếp --- ví dụ gắp vật, tránh chướng ngại vật, đặt đồ đúng vị trí.
    \item \textbf{Input}: quan sát hiện tại $o_t$ --- ảnh camera, vị trí/vận tốc khớp, vị trí vật cản hoặc mục tiêu.
    \item \textbf{Output}: hành động $a_t$ --- lực/vận tốc từng khớp, hoặc vị trí đích của tay gắp.
    \item \textbf{Bài toán chung}: cần một hàm quyết định $a_t = \pi(o_t)$ sao cho robot \emph{hoàn thành nhiệm vụ} \textbf{và} \emph{không gây nguy hiểm} (không đâm vào người/vật, không vượt giới hạn khớp).
\end{itemize}
\begin{alertblock}{Câu hỏi xuyên suốt lịch sử ngành}
Hàm quyết định $\pi$ này được \textbf{xây dựng như thế nào}? --- câu trả lời đã thay đổi qua nhiều giai đoạn (slide tiếp theo), và ShortStop nằm ở giai đoạn mới nhất.
\end{alertblock}
\end{frame}

\begin{frame}{Từ Classical Control đến Generative Policies}
\begin{itemize}
    \item \textbf{Control cổ điển} (PID, LQR): luật điều khiển suy từ phương trình vật lý $\Rightarrow$ \good{chứng minh được} an toàn/ổn định.
    \item \textbf{MPC}: giải bài toán tối ưu có ràng buộc mỗi bước (receding horizon) $\Rightarrow$ ràng buộc nằm ngay trong QP.
    \item \textbf{RL / imitation learning}: policy học từ dữ liệu $\Rightarrow$ \bad{mất khả năng chứng minh} --- policy là hộp đen.
    \item \textbf{Safe RL / CBF-QP}: cố lấy lại bảo đảm bằng barrier function, nhưng chỉ sửa \emph{một} hành động.
    \item \textbf{Generative policy (diffusion/flow)}: sinh cả \emph{chuỗi} hành động, không có certificate nào.
\end{itemize}
\vspace{4pt}
\footnotesize\textit{Phần tiếp theo trình bày chậm rãi từng giai đoạn --- bắt đầu từ MPC vì đây là nền tảng đã được trình bày trong nghiên cứu trước đó.}
\end{frame}

\begin{frame}{Model Predictive Control (MPC)}
\begin{itemize}
    \item \textbf{Ý tưởng}: tại mỗi bước thời gian, giải một bài toán tối ưu \emph{có ràng buộc}, dự đoán trước $N$ bước tương lai bằng mô hình động lực $f$, tìm ra cả chuỗi hành động tối ưu --- nhưng chỉ \textbf{thực thi hành động đầu tiên}, rồi giải lại từ đầu ở bước kế tiếp (\emph{receding horizon}).
    \item \textbf{Vì sao vẫn chứng minh được an toàn}: ràng buộc vật lý (giới hạn khớp, tránh vật cản) được đưa thẳng vào bài toán tối ưu dưới dạng \emph{constraint} --- nghiệm mà bộ giải (QP) trả về, nếu tồn tại, \textbf{luôn} thoả ràng buộc đó, không phải "hy vọng" mà là bảo đảm từ chính cấu trúc bài toán.
    \item MPC là đại diện tiêu biểu của nhóm phương pháp tối ưu có ràng buộc, đồng thời là nền tảng cho nhiều predictive safety filter hiện đại.
    \item \textbf{Giới hạn}: cần biết mô hình động lực $f$ khá chính xác, và bài toán tối ưu phải giải được \emph{real-time} mỗi bước.
\end{itemize}
\end{frame}

\begin{frame}{Reinforcement Learning \& Imitation Learning}
\begin{itemize}
    \item Thay vì viết công thức tay, để một mạng nơ-ron $\pi_\theta$ \textbf{học} luật điều khiển từ dữ liệu:
    \begin{itemize}
        \item \emph{Reinforcement Learning (RL)}: robot tự thử-sai trong môi trường/simulator, nhận thưởng khi làm đúng, phạt khi sai --- dần học ra chính sách tối ưu hoá thưởng.
        \item \emph{Imitation Learning}: cho robot xem hàng ngàn demo con người điều khiển, học \emph{bắt chước} trực tiếp, không cần reward.
    \end{itemize}
    \item \textbf{Cái mất}: $\pi_\theta$ giờ là một mạng nơ-ron hàng triệu tham số --- không còn phương trình tường minh để \textbf{chứng minh} nó luôn an toàn. Chỉ có thể \emph{kiểm định bằng thực nghiệm} (chạy thử nhiều lần xem có sai không), không phải bảo đảm toán học.
    \item \textbf{Cái được}: tổng quát hoá tốt hơn cho tác vụ khó viết tay (ví dụ cầm nắm vật mềm, thao tác khéo léo).
\end{itemize}
\end{frame}

\begin{frame}{Safe RL via Control Barrier Functions (CBF-QP)}
\begin{itemize}
    \item \textbf{Ý tưởng}: giữ nguyên policy học được (RL) làm việc chính, nhưng thêm một lớp ``gác cổng'' (safety filter) đứng giữa policy và robot thật.
    \item \textbf{Control Barrier Function (CBF)}: định nghĩa một hàm $h(x)$ sao cho ``an toàn'' $\Leftrightarrow$ $h(x)\ge 0$. Trước khi thực thi, giải một bài toán tối ưu nhỏ (QP) để sửa action đề xuất --- \textbf{sửa càng ít càng tốt} so với đề xuất gốc, miễn giữ được $h(x)\ge 0$.
    \item \textbf{Giới hạn}: (a) $h(x)$ phải \emph{thiết kế tay} cho từng tình huống nguy hiểm cụ thể; (b) chỉ sửa được \textbf{một} hành động tại một thời điểm, không nhìn xa hơn 1 bước.
    \item CBF-Shield là một baseline quan trọng vì cùng theo triết lý sử dụng lớp safety filter đặt ngoài policy.
\end{itemize}
\end{frame}

\begin{frame}{Generative Policies: Diffusion and Flow Matching}
\begin{itemize}
    \item Vẫn là policy học từ dữ liệu (như RL/Imitation Learning), nhưng thay vì trả về \textbf{một} hành động, sampler sinh ra cả một \alert{chuỗi} $H$ hành động tương lai (\emph{action chunk}), commit cùng lúc.
    \item Không có certificate nào đi kèm --- chunk được thực thi ngay khi sinh ra, khác hẳn MPC (ràng buộc trong bài toán tối ưu) và CBF-QP (lớp gác cổng sửa 1 action).
    \item Việc sinh cả action chunk giúp tạo quỹ đạo mượt và nhất quán theo thời gian, nhưng cũng làm phát sinh bài toán chứng nhận an toàn.
\end{itemize}
\end{frame}

\begin{frame}{Nguyên lý cốt lõi: vì sao cần ``sinh mẫu'' thay vì hồi quy}
\begin{itemize}
    \item \textbf{Hồi quy đơn giản thất bại} khi có nhiều lời giải đúng như nhau (multimodality) --- ví dụ vòng tay trái/phải qua vật cản $\to$ trung bình hoá 2 lời giải $\to$ đi thẳng vào vật cản, \emph{sai cả hai}.
    \item \textbf{Generative model giải quyết bằng cách nào}: học \emph{cả phân phối} lời giải hợp lý, mỗi lần dùng thì \emph{rút mẫu} một lời giải cụ thể --- giống mô hình sinh ảnh, hỏi ``vẽ con mèo'' 5 lần ra 5 con mèo khác nhau, đều hợp lý.
    \item \textbf{Khác biệt cốt lõi so với các phương pháp trước}: MPC/RL/CBF-QP đều trả về \emph{một} hành động xác định, có ràng buộc hoặc gác cổng đi kèm; generative policy trả về \emph{cả một chuỗi} $H$ hành động (action chunk), rút mẫu ngẫu nhiên, không ràng buộc, không certificate.
\end{itemize}
\end{frame}

\begin{frame}{Cơ chế: Diffusion Policy và Flow Matching sinh action chunk thế nào}
\begin{itemize}
    \item \textbf{Diffusion}: forward = làm nhiễu dần một mẫu thật; reverse = học khử nhiễu dần từ nhiễu trắng ra mẫu hợp lệ --- chia nhỏ thành nhiều bước, mỗi bước dễ học hơn học trực tiếp 1 bước.
    \item \textbf{Diffusion Policy} (Chi et al., RSS 2023): denoiser lấy quan sát $o_t$ làm điều kiện, khử nhiễu ra thẳng một \emph{action chunk} $H$ bước (không phải ảnh).
    \item \textbf{Flow matching} (Lipman et al., ICLR 2023): học một luồng liên tục/xác định (ODE) từ nhiễu tới mẫu --- ít bước tích hợp hơn diffusion, nhưng bản chất vẫn là sampler nhiều tham số.
    \item \textbf{Vì sao sinh cả chunk, không phải từng action}: mượt hơn, nhất quán theo thời gian, giảm số lần chạy mạng $\to$ tăng throughput --- nhưng đổi lại, chunk đã ``chốt'' là chốt cho cả $H$ bước, thực thi ngay khi sinh ra, không có bước kiểm tra nào ở giữa.
\end{itemize}
\end{frame}

\begin{frame}{Điểm mạnh / điểm yếu --- đây chính là motivation của ShortStop}
\begin{columns}[T]
\column{0.5\textwidth}
\textbf{\good{Điểm mạnh}}
\begin{itemize}
    \item Xử lý tốt multimodality (nhiều lời giải hợp lý)
    \item Action chunk mượt, nhất quán thời gian
    \item Throughput cao hơn (ít lần gọi mạng hơn)
    \item Tổng quát hoá tốt cho tác vụ khó viết tay (cầm nắm vật mềm, thao tác khéo léo)
\end{itemize}
\column{0.5\textwidth}
\textbf{\bad{Điểm yếu}}
\begin{itemize}
    \item Mạng nơ-ron cỡ chục--trăm triệu tham số --- hộp đen
    \item Phải qua nhiều bước nội tại (denoising/ODE) mới ra 1 chunk
    \item Ngẫu nhiên/đa modal --- hỏi lại có thể ra chunk khác
    \item Không certificate nào đi kèm khi chunk được thực thi
\end{itemize}
\end{columns}
\vspace{6pt}
\alert{$\Rightarrow$ chính 4 điểm yếu này là động lực (motivation) trực tiếp của ShortStop --- không phải nhận xét phụ.}
\end{frame}

\section{Part 1 --- Problem Formulation \& Core Insight}
%=====================================================================

\begin{frame}{Vấn đề cụ thể (Sec.~I của ShortStop)}
\begin{itemize}
    \item Diffusion/flow-matching policy $\to$ chuẩn mới cho robot manipulation, nhưng đi kèm đúng 4 điểm yếu ở trên.
    \item Quan sát lệch chuẩn (vật cản đặt sai, trễ tín hiệu, ánh sáng thay đổi) có thể đẩy sampler vào chế độ đâm vào bàn/người/giới hạn khớp --- và không ai biết trước, vì $\pi_\theta$ là hộp đen.
    \item Các tool neural-network reachability (Verisig, NNV, ReachNN,\dots) chỉ scale tới mạng nhỏ, horizon ngắn --- không tới nổi denoiser này qua contact dynamics.
    \item $\Rightarrow$ Full formal verification của $\pi_\theta$ $\to$ \bad{bất khả thi} --- vậy phải làm gì?
\end{itemize}
\end{frame}

\begin{frame}{Bất khả thi $\to$ khả thi: vì sao cần shield}
\begin{quote}
``We do not need to verify the policy to make it safe. We only need to verify, cheaply and just-in-time, the consequences of the chunk it is about to execute.''
\end{quote}
\begin{itemize}
    \item KHÔNG hỏi: ``$\pi_\theta$ luôn an toàn với mọi observation?'' (bất khả thi).
    \item CHỈ hỏi: ``với chunk cụ thể vừa đề xuất, hậu quả ngắn hạn có an toàn?'' (khả thi --- input đã cố định, chỉ còn 1 hệ động lực bị điều khiển).
    \item $\pi_\theta$ = black-box proposer, không phải đối tượng cần verify $\Rightarrow$ \alert{đây chính là lý do cần một shield đứng ngoài}.
\end{itemize}
\begin{center}
\Large \bad{Policy Safety?} $\;\Longrightarrow\;$ \good{Chunk Safety?}
\end{center}
\end{frame}

\begin{frame}{ShortStop = chuyển vai trò của control, không phát minh lại nó}
\begin{block}{Ý chính}
Công cụ control cổ điển (reachable set, invariant set) \textbf{không biến mất} khi chuyển sang policy học được --- nó chỉ đổi vai trò: từ \alert{controller} (MPC tự sinh action) sang \alert{supervisor} (ShortStop chỉ lọc/sửa đề xuất của một sampler mạnh hơn nhưng không giải thích được).
\end{block}
\begin{itemize}
    \item Ẩn dụ: một huấn luyện viên bay ngồi cạnh học viên --- không tự lái, chỉ nhìn trước vài giây và can thiệp khi thấy nguy hiểm cụ thể.
    \item Ngành đang đi từ policy nhỏ $\to$ foundation-model policy (VLA) $\Rightarrow$ câu hỏi ``ai bảo đảm an toàn'' ngày càng cấp thiết.
\end{itemize}
\end{frame}

\section{Part 2 --- Foundations of ShortStop}
%=====================================================================

\begin{frame}{Một runtime shield cần giải quyết những gì?}
\begin{itemize}
    \item Đứng giữa $\pi_\theta$ (black-box proposer) và robot thật, một shield ngắn hạn phải trả lời được 3 câu hỏi trước khi cho chunk đi qua:
    \begin{enumerate}
        \item \textbf{Dự đoán (Predict)}: nếu thực thi chunk này, robot sẽ ở đâu trong $H$ bước tới?
        \item \textbf{Đánh giá (Certify)}: quỹ đạo dự đoán đó có vi phạm vùng nguy hiểm $X_u$ hay không?
        \item \textbf{Sửa (Repair)}: nếu vi phạm, có cách nào chỉnh chunk ít nhất để tránh mà vẫn giữ tiến độ nhiệm vụ?
    \end{enumerate}
    \item Ba yêu cầu này dẫn trực tiếp tới kiến trúc của ShortStop và các công cụ lý thuyết mà hệ thống sử dụng.
\end{itemize}
\end{frame}

\begin{frame}{Related work}
\begin{center}
\includegraphics[width=0.5\linewidth]{images/shorstop_loop.png}
\end{center}

\footnotesize
\begin{itemize}
    \item \textbf{Predict} $\to$ \textbf{Reachability analysis} --- set propagation (CORA, Flow*), reachtube $\{\hat R_k\}$.
    \item \textbf{Certify} $\to$ \textbf{Signal Temporal Logic (STL)} --- robustness định lượng, tính trên cả tube, không chỉ 1 quỹ đạo.
    \item \textbf{Repair} $\to$ \textbf{CEGIS/CEGAR} --- dùng witness cụ thể để vừa bác bỏ vừa định hướng sửa.
\end{itemize}
\vspace{2pt}
\textbf{Không có khối nào mới} --- cái mới là \alert{cách ghép} chúng lại cho một generative sampler, lần đầu tiên.
\end{frame}

\section{Part 3 --- The P-R-C-S Framework}
%=====================================================================

\begin{frame}[t]{SHORTSTOP: vòng lặp P-R-C-S}
\small
\begin{block}{Tại mỗi decision step $t$}
SHORTSTOP giữ nguyên policy $\pi_\theta$, lấy một batch candidate chunks, kiểm tra an toàn và chỉ cho phép chunk phù hợp được thực thi.
\end{block}
\vspace{-0.2em}
\begin{description}
    \item[\textbf{P} Propose:] Lấy $K$ candidate từ policy: $\{a^{(i)}\}_{i=1}^{K}\sim\pi_\theta(\cdot\mid o_t)$.
    \item[\textbf{R} Reach:] Lan truyền over-approximation trong short horizon:
    \[
        \hat{\mathcal R}_{k+1}=\hat f(\hat{\mathcal R}_k,a_{t+k})\oplus\mathcal B(\epsilon_k+\bar w).
    \]
    \item[\textbf{C} Certify:] Tính STL robustness-to-go lower bound trên tube: $\underline{\rho}=\rho(\varphi,\hat{\mathcal R})$.
    \item[\textbf{S} Select/Repair:] Nếu chưa đạt margin, tìm counterexample, sửa trong trust region rồi chọn chunk có task progress tốt nhất.
\end{description}
\end{frame}

\begin{frame}[t,shrink=8]{P-R-C-S: từ candidate chunk đến action thực thi}
\footnotesize
\setlength{\abovedisplayskip}{2pt}
\setlength{\belowdisplayskip}{2pt}
\setlength{\abovedisplayshortskip}{1pt}
\setlength{\belowdisplayshortskip}{1pt}
\begin{block}{Một decision step tại thời điểm $t$}
Policy sinh một \textbf{batch nhỏ} candidate chunks; SHORTSTOP chỉ filter các đề xuất này thay vì tìm toàn bộ action space.
\end{block}
\vspace{-0.5em}
\begin{columns}[T,onlytextwidth]

%---------------- LEFT ----------------
\column{0.48\textwidth}

\textbf{1. Propose}

$\{a^{(i)}\}_{i=1}^{K}\sim\pi_\theta(\cdot\mid o_t),\;K=8$

\textbf{2. Reach + Certify}

Mỗi candidate được:

\begin{enumerate}
    \setlength{\itemsep}{0pt}
    \item[(R)] propagate thành reachable tube;
    \item[(C)] tính STL robustness lower bound.
\end{enumerate}

$\underline{\rho}\geq\varepsilon\Rightarrow\textbf{certified admissible}$

%---------------- RIGHT ----------------
\column{0.52\textwidth}

\textbf{3. Select / Repair}

Nếu candidate không đạt chứng nhận:

$\underline{\rho}<\varepsilon$

SHORTSTOP:

\begin{enumerate}
    \setlength{\itemsep}{0pt}
    \item tìm một \textbf{counterexample}
    trong reachable tube;
    \item dùng nó để xác định hướng repair;
    \item sửa candidate trong trust region và quay lại \textbf{Reach}.
\end{enumerate}
Nếu có candidate admissible: $a^*=\arg\max_{a^{(i)}\in\mathcal C} g(a^{(i)})$, \textbf{chỉ action đầu tiên được execute}.\\
Nếu tất cả candidate đều fail: chuyển sang \textbf{safe fallback controller}, sau đó quan sát lại và re-plan.

\end{columns}

\end{frame}

\section{Part 5 --- Literature, Gap \& Novelty}
%=====================================================================

\begin{frame}{Các hướng tiếp cận gần nhất trong literature}
\begin{itemize}
    \item Ứng với mỗi nhu cầu Predict/Certify/Repair (Part 2), ai đã làm rồi và còn thiếu gì? $\Rightarrow$ ưu tiên truy ngược đúng \textbf{3 baseline được benchmark trong Table~II}: \texttt{MPC-Filter}~[33], \texttt{CBF-Shield}~[38,39], \texttt{STL-Monitor}~[53].
    \item So với đúng bài baseline $\Rightarrow$ lập luận mạnh/yếu bám vào \emph{giới hạn phương pháp gốc}, không chỉ ``nghe liên quan''.
    \item \alert{Lưu ý}: số liệu Table~II/III hiện là số \textbf{kỳ vọng/placeholder} $\Rightarrow$ phần so sánh dưới đây bám \emph{giới hạn cấu trúc}, không bám chênh lệch số cụ thể.
\end{itemize}
\end{frame}

\begin{frame}{Baseline 1: MPC-Filter (Wabersich \& Zeilinger, 2021) --- nhu cầu Repair}
\textbf{Phương pháp gốc}: MPC filter sửa tối thiểu 1 action qua constrained QP --- chính là baseline \texttt{MPC-Filter}.
\begin{columns}[T]
\column{0.5\textwidth}
\textbf{\good{Mạnh hơn ShortStop}}
\begin{itemize}
    \item Lý thuyết recursive-feasibility trưởng thành
    \item Chỉ sửa 1 action --- đơn giản khi model lồi
\end{itemize}
\column{0.5\textwidth}
\textbf{\bad{Giới hạn cấu trúc (từ bài gốc)}}
\begin{itemize}
    \item Chỉ sửa 1 action, không có batch chunk $a_{t:t+H}$
    \item Cần model optimal-control lồi/tractable
    \item Không có STL / đặc tả thời gian phong phú
\end{itemize}
\end{columns}
\end{frame}

\begin{frame}{Baseline 2: CBF-Shield (Ames et al. 2017; Cheng et al. 2019) --- nhu cầu Certify+Repair tức thời}
\textbf{Phương pháp gốc}: QP tức thời ép $\dot h(x,u)\ge-\alpha(h(x))$ với $h(x)$ thiết kế tay --- chính là baseline \texttt{CBF-Shield}.
\begin{columns}[T]
\column{0.5\textwidth}
\textbf{\good{Mạnh hơn ShortStop}}
\begin{itemize}
    \item QP tuyến tính, giải cực nhanh
    \item Lý thuyết forward-invariance trưởng thành
\end{itemize}
\column{0.5\textwidth}
\textbf{\bad{Giới hạn cấu trúc (từ bài gốc)}}
\begin{itemize}
    \item $h(x)$ phải thiết kế tay từng hình dạng vùng nguy hiểm
    \item Chỉ sửa 1 input tức thời, không horizon $H{>}1$
    \item Cần $f,g$ tường minh, control-affine
    \item Không temporal-logic, không counterexample witness
\end{itemize}
\end{columns}
\end{frame}

\begin{frame}{Baseline 3: STL-Monitor (Deshmukh et al., 2017) --- nhu cầu Certify}
\textbf{Phương pháp gốc}: tính robustness STL trực tuyến trên 1 tín hiệu quan sát --- nguồn trực tiếp của Eq.~(2), và là baseline \texttt{STL-Monitor}.
\begin{columns}[T]
\column{0.5\textwidth}
\textbf{\good{Mạnh hơn ShortStop}}
\begin{itemize}
    \item Đệ quy hiệu quả, đã trưởng thành
    \item Tổng quát cho nhiều hệ cyber-physical
\end{itemize}
\column{0.5\textwidth}
\textbf{\bad{Giới hạn cấu trúc (từ bài gốc)}}
\begin{itemize}
    \item Monitor \emph{bị động} --- không reject-and-repair
    \item Tính trên 1 tín hiệu, không phải reachable set
    \item Không xử lý batch $K$ chunk, không counterexample
\end{itemize}
\end{columns}
\end{frame}

\begin{frame}{Từ 3 nhu cầu đến khoảng trống chung mà literature cũ chưa lấp được}
\begin{enumerate}
    \item Sửa/retrain generator --- \textbf{SafeDiffuser} (Xiao et al., ICLR 2025) làm vậy; ShortStop \textbf{không} retrain.
    \item Chỉ chứng nhận 1 action, không phải batch chunk --- \texttt{MPC-Filter}, \texttt{CBF-Shield}.
    \item Cần CBF thiết kế tay / model optimal-control tường minh --- \texttt{CBF-Shield}.
    \item Chỉ score 1 quỹ đạo nominal, không tube certificate, không counterexample --- \texttt{STL-Monitor}, và cả \textbf{MPM-STL} (Yu et al., 2024) dù đã đưa STL lên reachable set nhưng vẫn là monitor bị động.
\end{enumerate}
\end{frame}

\begin{frame}{Phần lớn các khối thành phần đã trưởng thành trong literature}
\begin{itemize}
    \item Set propagation interval/zonotope --- chuẩn CORA/Flow*.
    \item STL robustness đệ quy trên interval --- recursion tiêu chuẩn.
    \item MPC-Filter/CBF-Shield baseline --- tái dùng QP (Hildreth) đã có.
    \item Receding-horizon re-certification --- giống MPC cổ điển.
\end{itemize}
\begin{alertblock}{Vậy điều gì còn lại?}
Phần lớn các khối thành phần đều có sẵn trong literature; cái mới nằm ở cách kết hợp chúng.
\end{alertblock}
\end{frame}

\begin{frame}{Vậy ShortStop mới ở đâu? --- gói gọn trong 4 điểm}
\begin{enumerate}
    \item \textbf{Post-hoc, không retrain}: giữ nguyên $\pi_\theta$ đã đóng băng, không sửa/retrain generator như SafeDiffuser.
    \item \textbf{Reach + STL + CEGIS trong cùng một vòng per-decision} --- không chỉ dùng riêng lẻ 1 trong 3 như các baseline ở trên.
    \item \textbf{Chứng nhận cả action chunk (batch)}, không chỉ 1 action tức thời như MPC-Filter/CBF-Shield/STL-Monitor.
    \item \textbf{Đi kèm bộ guarantee lý thuyết} (bounded-time $\to$ all-time, Grönwall-type bound) --- không chỉ heuristic.
\end{enumerate}
\begin{alertblock}{So với công trình cùng thời}
\textbf{Path-Consistent Filtering} (Römer et al., arXiv:2511.06385, ICRA 2026) đồng thời cùng hướng nhưng chỉ nhắm diffusion, không thấy bounded-time/all-time theorem hay CEGIS tường minh trong mô tả công khai.
\end{alertblock}
\end{frame}

\section{Part 6 --- Guarantees \& Challenges}
%=====================================================================

\begin{frame}{Vì sao khó để có được các guarantee này?}
\begin{itemize}
    \item \textbf{Epsilon calibration}: calibrate $\varepsilon$ đúng theo phân phối thật, không phải một hằng số cố định.
    \item \textbf{Model error dưới distribution shift}: $\varepsilon$ sai dưới shift $\Rightarrow$ certificate sập âm thầm mà không cảnh báo.
    \item \textbf{Safe fallback}: xây $X_s,\pi_{fb}$ hợp lệ --- không tự động đúng, phải thiết kế riêng.
    \item \textbf{Safety vs.\ liveness}: 22\% reject không recover; certify \emph{batch} chunk từ hộp đen.
    \item \textbf{Real-time constraint}: giải bài toán đối nghịch đủ nhanh \emph{và} đúng, trong ngân sách $<$10ms cạnh GPU policy.
\end{itemize}
\vspace{4pt}
\textit{3 lát cắt khó xuyên suốt: khái niệm (đúng ranh giới tractable/intractable), toán học (đúng đắn dưới bất định mà không biết cấu trúc $\pi_\theta$), hệ thống ($<$10ms cạnh GPU policy).}
\end{frame}

\begin{frame}{Assumptions: điều kiện để guarantee đứng vững}
\begin{itemize}
    \item Mỗi challenge ở trên cần một giả định tương ứng để lý thuyết còn đứng vững:
    \item \textbf{Assumption 1}: reachable-set sound --- reachtube phải chứa quỹ đạo thật (đối phó với model error dưới distribution shift).
    \item \textbf{Assumption 2}: STL bound sound --- $\rho(\phi,\hat R)$ phải là cận dưới đúng của robustness thật (đối phó với sai số calibrate $\varepsilon$).
    \item \textbf{Assumption 3}: tồn tại safe fallback $(\pi_{fb}, X_s)$ --- luôn có đường lui an toàn (chính là challenge ``safe fallback'' ở trên).
    \item Mỗi assumption cũng bảo vệ đúng một mắt xích trong vòng P-R-C-S: (1)$\to$Reach, (2)$\to$Certify, (3)$\to$Select/Repair khi rỗng.
\end{itemize}
\end{frame}

\begin{frame}{Từ Bounded-time đến All-time safety}
\begin{itemize}
    \item \textbf{Theorem 1} (bounded-time safety): với 3 assumption trên, chunk được ShortStop chấp nhận không vi phạm $X_u$ trong $H$ bước tới.
    \item \textbf{Corollary 1} (all-time safety): quy nạp theo $t$ dựa trên $X_s$ bất biến --- an toàn từng bước cộng dồn thành an toàn suốt quỹ đạo.
\end{itemize}
\end{frame}

\begin{frame}{Conservatism vs. Horizon}
\begin{itemize}
    \item \textbf{Proposition 1} (Grönwall-type, Eq.~6--7): $r_k\le L^k r_0+(\varepsilon_{\max}+\bar w)\frac{L^k-1}{L-1}$, $\alpha(H)\le\alpha^\star+\beta(\epsilon+r_H)$.
    \item Đây chính là cơ sở toán học cho trực giác ``ngắn hạn vừa đủ vừa cần'' ở Part 3: $L^k$ tăng theo horizon $\Rightarrow$ tồn tại $H^\star$ tối ưu (thực nghiệm $H^\star=8$).
\end{itemize}
\end{frame}

\begin{frame}{Module 3: Method --- 4 khối cần học sâu (Sec.~III)}
\begin{itemize}
    \item \textbf{(R) Reachtube} (Eq.~1): $\hat R_{k+1} = \hat f(\hat R_k, a_{t+k}) \oplus B(\varepsilon_k+\bar w)$ --- cần học Minkowski sum $\oplus$, interval/zonotope, Lipschitz của $\hat f$.
    \item \textbf{(C) STL robustness-to-go} (Eq.~2): $\rho(\phi,\hat R) = \min_{0\le k\le H}\inf_{x\in\hat R_k}\mathrm{dist}(x,X_u)$ --- cần cú pháp STL cơ bản ($\square,\lozenge,\wedge,\vee$).
    \item \textbf{(S) CE search + repair} (Eq.~3--4): $\arg\min_{k,\,x\in\hat R_k}\mathrm{dist}(x,X_u)$, hướng sửa $d=\nabla_a\,\mathrm{dist}(\cdot)$, chiếu về trust region $\delta$ --- có họ hàng với QP-with-constraints đã làm (khác ở chỗ đây là gradient step có chiếu, không phải QP kín).
    \item \textbf{Kiểm tra mức độ hiểu}: trình bày lại Algorithm~1 từng bước mà không tham chiếu tài liệu gốc, trước khi sang Track B.
\end{itemize}
\vspace{4pt}
\footnotesize\textit{Ước lượng: 3--4 buổi --- module nặng nhất, đa số thời gian ở phần Reachtube/STL do khái niệm mới hoàn toàn.}
\end{frame}

\begin{frame}{Module 4: Lý thuyết an toàn (Sec.~IV + Phụ lục A)}
\begin{itemize}
    \item \textbf{Assumption 1--3}: reachable-set sound, STL bound sound, safe fallback ($\pi_{fb}, X_s$) tồn tại --- ghi rõ vì sao mỗi assumption cần thiết (theorem sập ở đâu nếu thiếu).
    \item \textbf{Theorem 1} (bounded-time safety) $\to$ \textbf{Corollary 1} (all-time safety, quy nạp theo $t$ dựa trên $X_s$ bất biến).
    \item \textbf{Proposition 1} (Grönwall-type, Eq.~6--7): $r_k\le L^k r_0+(\varepsilon_{\max}+\bar w)\frac{L^k-1}{L-1}$, $\alpha(H)\le\alpha^\star+\beta(\epsilon+r_H)$ --- giải thích vì sao có interior-optimum $H^\star=8$.
